    80→async function fetchPrediction(inputs: HealthInputs, geminiKey?: string): Promise<PredictionResult> {
    81→  const payload = { ...inputs } as Record<string, unknown>;
    82→  if (geminiKey) payload.gemini_api_key = geminiKey;
    83→  const res = await fetch(`${ML_API}/full-prediction`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) });
    84→  if (!res.ok) throw new Error(`API error: ${res.status}`);
    85→  return res.json();
    86→}
