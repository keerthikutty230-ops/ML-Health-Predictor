    46→def generate_synthetic_dataset(n_patients=500, seed=42):
    47→    rng = np.random.RandomState(seed)
    48→    ages = rng.normal(48, 14, n_patients).clip(18, 80).astype(int)
    49→    bmis = rng.normal(28.5, 6.5, n_patients).clip(16, 50).round(1)
    50→    bp_sys = rng.normal(130, 22, n_patients).clip(90, 200).astype(int)
    51→    bp_dia = rng.normal(85, 14, n_patients).clip(55, 130).astype(int)
    52→    glucose = rng.normal(115, 35, n_patients).clip(60, 250).astype(int)
    53→    cholesterol = rng.normal(210, 45, n_patients).clip(100, 400).astype(int)
    54→    heart_rate = rng.normal(75, 12, n_patients).clip(50, 120).astype(int)
    55→    insulin = rng.lognormal(3.5, 0.8, n_patients).clip(2, 500).round(1)
    56→
    57→    risk_score = np.zeros(n_patients)
    58→    risk_score += np.where(ages > 50, 0.15, 0.0) + np.where(ages > 65, 0.15, 0.0)
    59→    risk_score += np.where(bmis > 25, 0.1, 0.0) + np.where(bmis > 30, 0.15, 0.0)
    60→    risk_score += np.where(bp_sys > 130, 0.1, 0.0) + np.where(bp_sys > 140, 0.15, 0.0) + np.where(bp_dia > 90, 0.1, 0.0)
    61→    risk_score += np.where(glucose > 100, 0.15, 0.0) + np.where(glucose > 126, 0.2, 0.0)
    62→    risk_score += np.where(cholesterol > 200, 0.1, 0.0) + np.where(cholesterol > 240, 0.15, 0.0)
    63→    risk_score += rng.normal(0, 0.12, n_patients)
    64→    risk_score = risk_score.clip(0, 1)
    65→
    66→    def score_to_label(s):
    67→        if s < 0.33: return "low"
    68→        elif s < 0.60: return "moderate"
    69→        else: return "high"
    70→    risk_labels = [score_to_label(s) for s in risk_score]
    71→
    72→    def generate_diagnosis(risk, age, bmi, glucose, chol, sys_bp, dia_bp):
    73→        conditions = []
    74→        if risk == "high":
    75→            if glucose > 126: conditions.append("Type 2 Diabetes")
    76→            if chol > 240 or sys_bp > 140: conditions.append("Cardiovascular Disease")
    77→            if sys_bp > 140 and dia_bp > 90: conditions.append("Hypertension Stage 2")
    78→            if not conditions: conditions.append("Metabolic Syndrome")
    79→        elif risk == "moderate":
    80→            if glucose > 100: conditions.append("Prediabetes")
    81→            if sys_bp > 130: conditions.append("Elevated Blood Pressure")
    82→            if bmi > 30: conditions.append("Obesity")
    83→            if not conditions: conditions.append("Metabolic Risk Factors")
    84→        else:
    85→            conditions.append("Generally Healthy")
    86→        return ", ".join(conditions)
    87→
    88→    diagnoses = [generate_diagnosis(risk_labels[i], ages[i], bmis[i], glucose[i], cholesterol[i], bp_sys[i], bp_dia[i]) for i in range(n_patients)]
    89→
    90→    def generate_progression(risk):
    91→        if risk == "high":
    92→            return "Initial screening showed elevated markers. Follow-up at 6 months confirmed progressive risk. Treatment plan initiated with lifestyle modifications and medication."
    93→        elif risk == "moderate":
    94→            return "Routine checkup revealed borderline indicators. Referred for monitoring. 3-month follow-up showed stable condition with recommended dietary changes."
    95→        else:
    96→            return "Standard health screening within normal ranges. Annual checkup recommended. No immediate medical intervention required."
    97→    progressions = [generate_progression(r) for r in risk_labels]
    98→
    99→    return pd.DataFrame({
   100→        "patient_id": [f"P-{1000 + i}" for i in range(n_patients)],
   101→        "age": ages, "bmi": bmis, "blood_pressure_systolic": bp_sys, "blood_pressure_diastolic": bp_dia,
   102→        "glucose": glucose, "cholesterol": cholesterol, "heart_rate": heart_rate, "insulin": insulin,
   103→        "risk_level": risk_labels, "risk_score": risk_score.round(3),
   104→        "diagnosis": diagnoses, "progression_notes": progressions,
   105→    })

   222→# 4. MODEL TRAINING
   223→# ========================================================================
   224→def train_model(df):
   225→    global model, scaler, nn_model, df_patients
   226→    df_patients = df
   227→    X = df[FEATURE_COLS].values
   228→    label_map = {"low": 0, "moderate": 1, "high": 2}
   229→    y = df["risk_level"].map(label_map).values
   230→    scaler = StandardScaler()
   231→    X_scaled = scaler.fit_transform(X)
   232→    model = GradientBoostingClassifier(n_estimators=150, max_depth=5, learning_rate=0.1, min_samples_split=10, min_samples_leaf=5, subsample=0.8, random_state=42)
   233→    model.fit(X_scaled, y)
   234→    logger.info(f"Model trained. Accuracy: {model.score(X_scaled, y):.4f}")
   235→    nn_model = NearestNeighbors(n_neighbors=3, metric="cosine", algorithm="brute")
   236→    nn_model.fit(X_scaled)

   294→@app.post("/api/full-prediction")
   295→async def full_prediction(input_data: HealthInput):
   296→    if model is None or scaler is None or nn_model is None or df_patients is None or df_hospitals is None:
   297→        raise HTTPException(status_code=503, detail="Model not initialized")
   298→
   299→    # 1. Risk Prediction
   300→    user_vector = np.array([[input_data.age, input_data.bmi, input_data.blood_pressure_systolic, input_data.blood_pressure_diastolic, input_data.glucose, input_data.cholesterol, input_data.heart_rate, input_data.insulin]])
   301→    user_scaled = scaler.transform(user_vector)
   302→    prediction = model.predict(user_scaled)[0]
   303→    probabilities = model.predict_proba(user_scaled)[0]
   304→    inv_map = {0: "low", 1: "moderate", 2: "high"}
   305→    predicted_tier = inv_map[prediction]
   306→    importance = dict(sorted(zip(FEATURE_COLS, model.feature_importances_.round(4)), key=lambda x: x[1], reverse=True))
   307→    risk_result = {
   308→        "risk_tier": predicted_tier, "risk_score": round(float(probabilities[prediction]), 4),
   309→        "risk_probability": {"low": round(float(probabilities[0]), 4), "moderate": round(float(probabilities[1]), 4), "high": round(float(probabilities[2]), 4)},
   310→        "feature_importance": importance
   311→    }
   312→
   313→    # 2. Historical Similarity Matching
   314→    distances, indices = nn_model.kneighbors(user_scaled, n_neighbors=3)
   315→    matches = []
   316→    seen = set()
   317→    for i in range(len(indices[0])):
   318→        idx = indices[0][i]
   319→        patient = df_patients.iloc[idx]
   320→        similarity = round(1.0 - distances[0][i], 4)
   321→        if patient["patient_id"] in seen: continue
   322→        seen.add(patient["patient_id"])
   323→        user_traits = {"Age": input_data.age, "Bmi": input_data.bmi, "Bp Sys": input_data.blood_pressure_systolic, "Glucose": input_data.glucose, "Cholesterol": input_data.cholesterol}
   324→        pat_traits = {"Age": patient["age"], "Bmi": patient["bmi"], "Bp Sys": patient["blood_pressure_systolic"], "Glucose": patient["glucose"], "Cholesterol": patient["cholesterol"]}
   325→        overlapping = []
   326→        for key in user_traits:
   327→            u, p = user_traits[key], float(pat_traits[key])
   328→            diff_pct = abs(u - p) / max(u, p, 1) * 100
   329→            if diff_pct < 15:
   330→                overlapping.append({"trait": key.replace(" ", " ").title(), "user_value": float(u), "patient_value": round(p, 1), "deviation_pct": round(diff_pct, 1)})
   331→        matches.append({"patient_id": patient["patient_id"], "similarity_score": similarity, "profile": {"age": int(patient["age"]), "bmi": float(patient["bmi"]), "blood_pressure": f"{int(patient['blood_pressure_systolic'])}/{int(patient['blood_pressure_diastolic'])}", "glucose": int(patient["glucose"]), "cholesterol": int(patient["cholesterol"]), "heart_rate": int(patient["heart_rate"])}, "risk_level": patient["risk_level"], "diagnosis": patient["diagnosis"], "overlapping_traits": overlapping, "historical_progression": patient["progression_notes"]})
   332→        if len(matches) >= 2: break
   333→
   334→    # 3. Location-Aware Hospital Recommendation
   335→    city_filter = input_data.city.strip().title()
   336→    state_filter = input_data.state.strip().upper()
   337→    local_hospitals = df_hospitals[
   338→        (df_hospitals["city"].str.lower() == city_filter.lower()) &
   339→        (df_hospitals["state"].str.upper() == state_filter.upper())
   340→    ]
   341→    if len(local_hospitals) == 0:
   342→        local_hospitals = df_hospitals[df_hospitals["state"].str.upper() == state_filter.upper()]
   343→    if len(local_hospitals) == 0:
   344→        local_hospitals = df_hospitals
   345→
   346→    required_specialties = ["Internal Medicine"]
   347→    if input_data.blood_pressure_systolic > 130 or input_data.cholesterol > 200: required_specialties.append("Cardiology")
   348→    if input_data.glucose > 100 or input_data.bmi > 30: required_specialties.append("Endocrinology")
   349→    if input_data.glucose > 126 and input_data.blood_pressure_systolic > 140: required_specialties.append("Nephrology")
   350→
   351→    hospital_results = []
   352→    for _, h in local_hospitals.iterrows():
   353→        h_specs = h["specialties"]
   354→        spec_matches = len(set(required_specialties) & set(h_specs))
   355→        spec_score = spec_matches / len(required_specialties) if required_specialties else 0
   356→        dist_score = max(0, 1 - h["distance_km"] / 20)
   357→        rating_score = h["rating"] / 5.0
   358→        if predicted_tier == "high":
   359→            combined = spec_score * 0.45 + rating_score * 0.35 + dist_score * 0.20
   360→        else:
   361→            combined = spec_score * 0.35 + rating_score * 0.25 + dist_score * 0.40
   362→        hospital_results.append({"id": h["id"], "name": h["name"], "city": h["city"], "state": h["state"], "matched_specialties": list(set(required_specialties) & set(h_specs)), "all_specialties": h_specs, "rating": h["rating"], "distance_km": h["distance_km"], "address": h["address"], "phone": h["phone"], "beds": int(h["beds"]), "emergency": h["emergency"], "doctors": h.get("doctors", []), "match_score": round(combined, 4)})
   363→    hospital_results.sort(key=lambda x: x["match_score"], reverse=True)
   364→
   365→    # 4. Conditional Care Guidance
   366→    care_guidance = generate_care_guidance(predicted_tier, input_data)
   367→
   368→    # 5. User profile for comparative display
   369→    user_profile = {"age": input_data.age, "bmi": input_data.bmi, "blood_pressure": f"{input_data.blood_pressure_systolic}/{input_data.blood_pressure_diastolic}", "glucose": input_data.glucose, "cholesterol": input_data.cholesterol, "heart_rate": input_data.heart_rate, "insulin": input_data.insulin, "location": f"{input_data.city}, {input_data.state}"}
   370→
   371→    # 6. Gemini AI Summary
   372→    ai_summary = None
   373→    if input_data.gemini_api_key:
   374→        try:
   375→            import google.generativeai as genai
   376→            genai.configure(api_key=input_data.gemini_api_key)
   377→            m = genai.GenerativeModel("gemini-2.0-flash")
   378→            prompt = f"""You are a compassionate, professional medical AI assistant. Generate a clear executive health summary.
   379→**IMPORTANT**: NOT a medical diagnosis. Always recommend consulting a healthcare professional.
   380→
   381→## User: {input_data.age}yo, BMI={input_data.bmi}, BP={input_data.blood_pressure_systolic}/{input_data.blood_pressure_diastolic}, Glucose={input_data.glucose}, Cholesterol={input_data.cholesterol}, Location={input_data.city}, {input_data.state}
   382→
   383→## Risk: {predicted_tier.upper()} ({risk_result['risk_score']*100:.1f}% confidence)
   384→Probs: Low={probabilities[0]*100:.1f}% Moderate={probabilities[1]*100:.1f}% High={probabilities[2]*100:.1f}%
   385→Top factors: {', '.join(list(importance.keys())[:3])}
   386→
   387→## Matching Patients:
   388→{chr(10).join([f'- {m["patient_id"]} ({m["similarity_score"]*100:.1f}% match): {m["diagnosis"]}. Progression: {m["historical_progression"]}' for m in matches])}
   389→
   390→## Local Hospitals ({input_data.city}, {input_data.state}):
   391→{chr(10).join([f'- {h["name"]}: {h["rating"]}/5, {h["distance_km"]}km, Specialists: {', '.join(h["doctors"])}' for h in hospital_results[:3]])}

   393→## Care Guidance: {care_guidance['message']}
   394→
   395→Provide a compassionate, professional summary with: 1) Brief assessment, 2) Key risk factors explained, 3) 3-5 actionable next steps, 4) Hospital guidance for {input_data.city}. Keep tone empathetic and non-alarmist."""
   396→            response = await m.generate_content_async(prompt)
   397→            ai_summary = response.text
   398→        except Exception as e:
   399→            logger.error(f"Gemini error: {e}")

   401→    return FullPredictionResponse(risk=risk_result, historical_matches=matches, recommended_hospitals=hospital_results, care_guidance=care_guidance, user_profile=user_profile, ai_summary=ai_summary)
